Remarks

A shader may declare any of these uniforms and ShaderView sets only the ones it declares: iTime in seconds; iResolution, iPointer, and iClick in control coordinates; and iProgress and iPress from zero to one. Progress and press can be driven with Nova springs.

AuroraRipple

Apple-Intelligence-style effect: aurora background, pointer glow, and an expanding ripple ring driven by iProgress (animate it with a Nova spring).

public const string AuroraRipple = "uniform float iTime;\nuniform float2 iResolution;\nuniform float2 iPointer;\nuniform float iProgress;\n\nhalf4 main(float2 fragCoord) {\n    float2 uv = fragCoord / iResolution;\n    float aspect = iResolution.x / max(iResolution.y, 1.0);\n    float2 pos = float2(uv.x * aspect, uv.y);\n    float2 ptr = float2(iPointer.x / iResolution.x * aspect, iPointer.y / iResolution.y);\n\n    float d = distance(pos, ptr);\n\n    float t = iTime * 0.35;\n    half3 color = half3(\n        half(0.16 + 0.10 * sin(6.28318 * (uv.x * 0.7 + t))),\n        half(0.10 + 0.09 * sin(6.28318 * (uv.y * 0.6 + t + 0.33))),\n        half(0.26 + 0.14 * sin(6.28318 * ((uv.x + uv.y) * 0.5 + t + 0.66))));\n\n    color += half3(0.30, 0.18, 0.52) * half(exp(-d * 6.0));\n\n    // Swell in quickly, then fade out before the end so nothing is left frozen on screen.\n    // Written with clamp rather than smoothstep, which this SkSL dialect does not provide.\n    float rise = clamp(iProgress / 0.12, 0.0, 1.0);\n    float fall = clamp((1.0 - iProgress) / 0.45, 0.0, 1.0);\n    float envelope = rise * rise * (3.0 - 2.0 * rise) * fall * fall * (3.0 - 2.0 * fall);\n\n    float radius = iProgress * 1.6;\n    float ring = exp(-pow((d - radius) * 9.0, 2.0));\n    float wave = sin(d * 45.0 - iProgress * 22.0) * exp(-d * 2.5);\n    color += half3(0.85, 0.80, 1.00) * half(ring * 0.9 * envelope);\n    color += half3(half(wave * 0.18 * envelope));\n\n    return half4(color, 1.0);\n}"

BackdropBlur

A live backdrop blur without the tint, for dimming content behind a panel.

public const string BackdropBlur = "uniform shader iContent;\nuniform float2 iResolution;\nuniform float2 iOffset;\nuniform float iProgress;\n\nhalf4 main(float2 fragCoord) {\n    float2 source = fragCoord + iOffset;\n    float radius = 1.0 + iProgress * 14.0;\n\n    half4 sum = iContent.eval(source) * 4.0;\n    sum += iContent.eval(source + float2( radius, 0)) * 2.0;\n    sum += iContent.eval(source + float2(-radius, 0)) * 2.0;\n    sum += iContent.eval(source + float2(0,  radius)) * 2.0;\n    sum += iContent.eval(source + float2(0, -radius)) * 2.0;\n    sum += iContent.eval(source + float2( radius,  radius) * 0.7);\n    sum += iContent.eval(source + float2(-radius,  radius) * 0.7);\n    sum += iContent.eval(source + float2( radius, -radius) * 0.7);\n    sum += iContent.eval(source + float2(-radius, -radius) * 0.7);\n\n    return sum / 16.0;\n}"

ClickRipple

A ripple starting exactly where the pointer went down. Animate iProgress 0 to 1 per press; iClick holds the origin still while it expands.

public const string ClickRipple = "uniform float2 iResolution;\nuniform float2 iClick;\nuniform float iProgress;\n\nhalf4 main(float2 fragCoord) {\n    float2 uv = fragCoord / iResolution;\n    float aspect = iResolution.x / max(iResolution.y, 1.0);\n    float2 pos = float2(uv.x * aspect, uv.y);\n    float2 origin = float2(iClick.x / iResolution.x * aspect, iClick.y / iResolution.y);\n\n    float d = distance(pos, origin);\n    float radius = iProgress * 1.9;\n\n    // Fade both ends to avoid a persistent point at progress zero.\n    float rise = clamp(iProgress / 0.06, 0.0, 1.0);\n    float fall = clamp((1.0 - iProgress) / 0.5, 0.0, 1.0);\n    float envelope = rise * fall * fall * (3.0 - 2.0 * fall);\n\n    float disc = clamp((radius - d) * 12.0, 0.0, 1.0);\n    float edge = exp(-pow((d - radius) * 14.0, 2.0));\n\n    half3 base = half3(0.09, 0.10, 0.15);\n    half3 fill = half3(0.42, 0.36, 0.95);\n    half3 color = mix(base, fill, half(disc * envelope * 0.85));\n    color += half3(0.85, 0.82, 1.00) * half(edge * envelope * 0.7);\n    return half4(color, 1.0);\n}"

Dissolve

Transition effects sample a captured frame of real UI through iContent. Use them with ShaderTransition, which supplies the snapshot.

public const string Dissolve = "uniform shader iContent;\nuniform float2 iResolution;\nuniform float iProgress;\n\nfloat noise(float2 p) {\n    return fract(sin(dot(p, float2(12.9898, 78.233))) * 43758.5453);\n}\n\nhalf4 main(float2 fragCoord) {\n    half4 source = iContent.eval(fragCoord);\n\n    // Per-pixel thresholds produce a grain dissolve.\n    float threshold = noise(floor(fragCoord / 3.0));\n    float alpha = clamp((1.0 - iProgress) * 1.35 - threshold * 0.35, 0.0, 1.0);\n    return source * half(alpha);\n}"

Foil

Holographic foil whose hue shifts with the pointer, like a trading card.

public const string Foil = "uniform float iTime;\nuniform float2 iResolution;\nuniform float2 iPointer;\n\nhalf3 hue(float h) {\n    float3 k = float3(1.0, 0.6666667, 0.3333333);\n    float3 p = abs(fract(float3(h, h, h) + k) * 6.0 - 3.0);\n    return half3(clamp(p - 1.0, 0.0, 1.0));\n}\n\nhalf4 main(float2 fragCoord) {\n    float2 uv = fragCoord / iResolution;\n    float2 ptr = iPointer / iResolution;\n\n    float2 delta = uv - ptr;\n    float dist = length(delta);\n\n    // Two-argument atan is not available here, so the hue is driven by the direction\n    // components directly. It gives the same tilt-to-shift feel without the angle.\n    float tilt = (delta.x - delta.y) * 0.5;\n\n    float bands = sin((uv.x + uv.y) * 14.0 + iTime * 0.6) * 0.5 + 0.5;\n    float h = fract(tilt + dist * 0.9 + bands * 0.15 + iTime * 0.03);\n\n    half3 color = hue(h) * half(0.55 + 0.45 * bands);\n    color *= half(0.35 + 0.75 * exp(-dist * 1.6));\n    return half4(color, 1.0);\n}"

Glass

Creates a frosted-glass effect over live content.

public const string Glass = "uniform shader iContent;\nuniform float2 iResolution;\nuniform float2 iOffset;\nuniform float iProgress;\n\nhalf4 main(float2 fragCoord) {\n    float2 source = fragCoord + iOffset;\n    float radius = 3.0 + iProgress * 11.0;\n\n    // A weighted 3x3 kernel avoids visible duplicate silhouettes without adding samples.\n    half4 sum = iContent.eval(source) * 4.0;\n    sum += iContent.eval(source + float2( radius, 0)) * 2.0;\n    sum += iContent.eval(source + float2(-radius, 0)) * 2.0;\n    sum += iContent.eval(source + float2(0,  radius)) * 2.0;\n    sum += iContent.eval(source + float2(0, -radius)) * 2.0;\n    sum += iContent.eval(source + float2( radius,  radius) * 0.7);\n    sum += iContent.eval(source + float2(-radius,  radius) * 0.7);\n    sum += iContent.eval(source + float2( radius, -radius) * 0.7);\n    sum += iContent.eval(source + float2(-radius, -radius) * 0.7);\n\n    half4 blurred = sum / 16.0;\n\n    // Lift and desaturate slightly so text stays readable on top.\n    half3 tinted = mix(blurred.rgb, half3(1.0, 1.0, 1.0), half(0.18));\n\n    float2 uv = fragCoord / iResolution;\n    float sheen = exp(-uv.y * 5.0) * 0.16;\n    return half4(tinted + half3(half(sheen)), 1.0);\n}"

MeshGradient

Slowly drifting mesh gradient, for hero or card backgrounds.

public const string MeshGradient = "uniform float iTime;\nuniform float2 iResolution;\n\nhalf4 main(float2 fragCoord) {\n    float2 uv = fragCoord / iResolution;\n    float t = iTime * 0.18;\n\n    float2 a = float2(0.25 + 0.18 * sin(t * 1.1), 0.30 + 0.16 * cos(t * 0.9));\n    float2 b = float2(0.75 + 0.16 * cos(t * 0.8), 0.28 + 0.18 * sin(t * 1.3));\n    float2 c = float2(0.40 + 0.20 * sin(t * 0.7), 0.78 + 0.14 * cos(t * 1.1));\n\n    float wa = exp(-distance(uv, a) * 3.2);\n    float wb = exp(-distance(uv, b) * 3.2);\n    float wc = exp(-distance(uv, c) * 3.2);\n    float sum = max(wa + wb + wc, 0.0001);\n\n    half3 color = (half3(0.45, 0.36, 0.98) * half(wa)\n                 + half3(0.95, 0.42, 0.71) * half(wb)\n                 + half3(0.24, 0.78, 0.72) * half(wc)) / half(sum);\n\n    return half4(color, 1.0);\n}"

Shimmer

Skeleton shimmer: a highlight sweeping across a loading placeholder. It loops on iTime, so it needs no animation state of its own.

public const string Shimmer = "uniform float iTime;\nuniform float2 iResolution;\n\nhalf4 main(float2 fragCoord) {\n    float2 uv = fragCoord / iResolution;\n\n    // The sweep runs on a diagonal and repeats with a pause between passes.\n    float cycle = fract(iTime * 0.6);\n    float head = cycle * 2.2 - 0.6;\n    float band = (uv.x + uv.y * 0.35) - head;\n    float sweep = exp(-band * band * 26.0);\n\n    float alpha = 0.07 + 0.22 * sweep;\n    return half4(half3(1.0) * half(alpha), half(alpha));\n}"

Wipe

A diagonal wipe with a bright leading edge, the classic curtain transition.

public const string Wipe = "uniform shader iContent;\nuniform float2 iResolution;\nuniform float iProgress;\n\nhalf4 main(float2 fragCoord) {\n    float2 uv = fragCoord / iResolution;\n    half4 source = iContent.eval(fragCoord);\n\n    float edge = (uv.x * 0.7 + uv.y * 0.3);\n    float front = iProgress * 1.35 - 0.15;\n    float alpha = clamp((edge - front) * 14.0, 0.0, 1.0);\n\n    // Highlight the wipe boundary.\n    float glow = exp(-pow((edge - front) * 16.0, 2.0)) * (1.0 - iProgress);\n    return source * half(alpha) + half4(0.55, 0.50, 1.0, 0.0) * half(glow * 0.5);\n}"
